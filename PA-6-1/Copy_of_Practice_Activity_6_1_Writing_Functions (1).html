def plot_bills_dynamic(species=None, island=None):
  df = penguins

  if species is not None:
    df = df[df["species"] == species]
  if island is not None:
    df = df[df["island"] == island]

  return (
        ggplot(df, aes(x="bill_length_mm", y="bill_depth_mm"))
        + geom_point()
    )
